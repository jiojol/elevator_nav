cmake_minimum_required(VERSION 3.16)
project(go2_local_simulation LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_BUILD_TYPE Release)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${OUTPUT_PREFIX}/lib")
set(CMAKE_BUILD_RPATH "${OUTPUT_PREFIX}/lib;${ROS_PREFIX}/lib;${DEPENDENCIES}/tf/lib")
find_package(Eigen3 REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs)
find_package(PCL REQUIRED COMPONENTS common io filters registration kdtree search)
find_package(OpenMP REQUIRED)
set(SCAN "${REPOSITORY}/nav/src/prior_local_planner/src/planner")
set(CV_BRIDGE "${DEPENDENCIES}/vision_opencv/cv_bridge")
include_directories(SYSTEM "${ROS_PREFIX}/include" "${DEPENDENCIES}/tf/include"
  "${DEPENDENCIES}/pcl_conversions/include" ${EIGEN3_INCLUDE_DIRS} ${PCL_INCLUDE_DIRS}
  ${OpenCV_INCLUDE_DIRS} "${CV_BRIDGE}/include")
include_directories("${OUTPUT_PREFIX}/include")
include_directories("${REPOSITORY}/nav/src/dog_nav_bringup/include")
foreach(package plan_env path_searching bspline_opt traj_utils plan_manage)
  include_directories("${SCAN}/${package}/include")
endforeach()
set(ROS_LIBRARIES)
foreach(name roscpp rosconsole rosconsole_log4cxx rosconsole_backend_interface
    roscpp_serialization rostime xmlrpcpp cpp_common tf2 tf2_ros message_filters actionlib)
  find_library(ROS_${name} NAMES ${name} PATHS "${ROS_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
  list(APPEND ROS_LIBRARIES "${ROS_${name}}")
endforeach()
list(APPEND ROS_LIBRARIES "${DEPENDENCIES}/tf/lib/libtf.so")
# Keep the ROS ABI from this underlay; PCL/OpenCV are rebuilt against the host.
add_compile_definitions(ROS_PACKAGE_NAME="scan_planner" ROSCONSOLE_BACKEND_LOG4CXX)
# Noetic vendors an old Boost endian header. Use the installed host version
# when compiling against Ubuntu 24.04; retain the upstream conversion code.
file(READ "${CV_BRIDGE}/src/cv_bridge.cpp" CV_BRIDGE_SOURCE)
string(REPLACE "#include \"boost/endian/conversion.hpp\"" "#include <boost/endian/conversion.hpp>" CV_BRIDGE_SOURCE "${CV_BRIDGE_SOURCE}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/cv_bridge_host.cpp" "${CV_BRIDGE_SOURCE}")
add_library(cv_bridge SHARED "${CMAKE_CURRENT_BINARY_DIR}/cv_bridge_host.cpp" "${CV_BRIDGE}/src/rgb_colors.cpp")
target_link_libraries(cv_bridge ${ROS_LIBRARIES} ${OpenCV_LIBRARIES})
add_library(plan_env SHARED "${SCAN}/plan_env/src/grid_map.cpp" "${SCAN}/plan_env/src/raycast.cpp")
target_link_libraries(plan_env cv_bridge ${ROS_LIBRARIES} ${PCL_LIBRARIES})
add_library(path_searching SHARED "${SCAN}/path_searching/src/dyn_a_star.cpp")
target_link_libraries(path_searching plan_env)
add_library(bspline_opt SHARED "${SCAN}/bspline_opt/src/uniform_bspline.cpp"
  "${SCAN}/bspline_opt/src/bspline_optimizer.cpp" "${SCAN}/bspline_opt/src/gradient_descent_optimizer.cpp")
target_link_libraries(bspline_opt path_searching)
# Match the retained native build's treatment of legacy unused gradient limits.
target_compile_options(bspline_opt PRIVATE -Wno-narrowing)
add_library(traj_utils SHARED "${SCAN}/traj_utils/src/planning_visualization.cpp" "${SCAN}/traj_utils/src/polynomial_traj.cpp")
target_link_libraries(traj_utils bspline_opt)
add_executable(scan_planner_node "${SCAN}/plan_manage/src/scan_planner_node.cpp"
  "${SCAN}/plan_manage/src/scan_replan_fsm.cpp" "${SCAN}/plan_manage/src/planner_manager.cpp")
target_link_libraries(scan_planner_node traj_utils)
add_executable(closed_loop_controller "${SCAN}/plan_manage/src/closed_loop_controller.cpp")
target_link_libraries(closed_loop_controller bspline_opt)
set_target_properties(scan_planner_node closed_loop_controller PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${OUTPUT_PREFIX}/lib/scan_planner")
add_executable(reference_path_transform "${REPOSITORY}/nav/src/dog_nav_bringup/src/reference_path_transform.cpp")
target_link_libraries(reference_path_transform ${ROS_LIBRARIES})
set_target_properties(reference_path_transform PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${OUTPUT_PREFIX}/lib/dog_nav_bringup")
add_library(local_fast_gicp SHARED
  "${REPOSITORY}/nav/src/fast_gicp/src/fast_gicp/gicp/lsq_registration.cpp"
  "${REPOSITORY}/nav/src/fast_gicp/src/fast_gicp/gicp/fast_gicp.cpp")
target_include_directories(local_fast_gicp PUBLIC "${REPOSITORY}/nav/src/fast_gicp/include")
target_link_libraries(local_fast_gicp ${PCL_LIBRARIES} OpenMP::OpenMP_CXX)
add_executable(fastlio_map_localizer "${REPOSITORY}/nav/src/dog_nav_bringup/src/fastlio_map_localizer.cpp")
target_link_libraries(fastlio_map_localizer local_fast_gicp ${ROS_LIBRARIES} ${PCL_LIBRARIES})
set_target_properties(fastlio_map_localizer PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${OUTPUT_PREFIX}/lib/dog_nav_bringup")
add_executable(scan_spatial_guard_test "${REPOSITORY}/system/tests/scan_spatial_guard_test.cpp"
  "${SCAN}/plan_manage/src/planner_manager.cpp")
target_link_libraries(scan_spatial_guard_test traj_utils)
set_target_properties(scan_spatial_guard_test PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${OUTPUT_PREFIX}/lib/scan_planner")
add_executable(body_reference_collision_test "${REPOSITORY}/system/tests/body_reference_collision_test.cpp")
target_link_libraries(body_reference_collision_test plan_env)
set_target_properties(body_reference_collision_test PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${OUTPUT_PREFIX}/lib/scan_planner")
add_executable(narrow_prior_replan_test "${REPOSITORY}/system/tests/narrow_prior_replan_test.cpp"
  "${SCAN}/plan_manage/src/planner_manager.cpp")
target_link_libraries(narrow_prior_replan_test traj_utils)
set_target_properties(narrow_prior_replan_test PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${OUTPUT_PREFIX}/lib/scan_planner")
add_executable(astar_endpoint_regression_test "${REPOSITORY}/system/tests/astar_endpoint_regression_test.cpp")
target_link_libraries(astar_endpoint_regression_test bspline_opt)
set_target_properties(astar_endpoint_regression_test PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${OUTPUT_PREFIX}/lib/scan_planner")

add_executable(map_publication_test "${REPOSITORY}/system/tests/map_publication_test.cpp")
target_link_libraries(map_publication_test plan_env)
set_target_properties(map_publication_test PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${OUTPUT_PREFIX}/lib/scan_planner")
